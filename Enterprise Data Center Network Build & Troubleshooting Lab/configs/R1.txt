! =====================================================================
! DOH-DC01-R1  |  Cisco 2911  |  Inter-VLAN gateway (router-on-a-stick)
! Role: default gateway for VLAN 10/20/99, DHCP server for clients
! =====================================================================
enable
configure terminal
hostname DOH-DC01-R1
no ip domain-lookup
service password-encryption
enable secret DcLab@2026
banner motd # AUTHORIZED ACCESS ONLY - DOHA-DC01 - All activity is logged #

! ---------- Physical trunk interface ----------
interface gigabitEthernet0/0
 description TRUNK-TO-DOH-DC01-SW1 | G0/1 | CAB-0001
 no ip address
 no shutdown
exit

! ---------- Sub-interfaces (one per VLAN) ----------
interface gigabitEthernet0/0.10
 description GW-VLAN10-SERVERS
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
exit
interface gigabitEthernet0/0.20
 description GW-VLAN20-CLIENTS
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
exit
interface gigabitEthernet0/0.99
 description GW-VLAN99-MGMT
 encapsulation dot1Q 99
 ip address 192.168.99.1 255.255.255.0
exit
interface gigabitEthernet0/0.999
 description NATIVE-VLAN-NO-IP
 encapsulation dot1Q 999 native
exit

! ---------- DHCP for client VLAN 20 ----------
ip dhcp excluded-address 192.168.20.1 192.168.20.99
ip dhcp pool CLIENTS-VLAN20
 network 192.168.20.0 255.255.255.0
 default-router 192.168.20.1
 dns-server 192.168.10.10
exit

! ---------- SSH ----------
ip domain-name doha-dc01.lab
username netadmin privilege 15 secret DcLab@2026
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line console 0
 logging synchronous
 exec-timeout 10 0
exit
line vty 0 4
 login local
 transport input ssh
 exec-timeout 10 0
exit

! ---------- Monitoring (Phase 6) ----------
service timestamps log datetime msec
logging host 192.168.10.10
ntp server 192.168.10.10

end
write memory
